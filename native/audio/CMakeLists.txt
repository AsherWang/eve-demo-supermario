cmake_minimum_required(VERSION 3.21)
project(MarioNativeAudio LANGUAGES CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "The native playback backend currently requires Windows")
endif()
find_package(EVEngine CONFIG REQUIRED)
add_eve_plugin(mario_audio SOURCES NativeAudio.cpp Mixer.cpp LIBRARIES winmm)
target_compile_definitions(mario_audio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(mario_audio PRIVATE /W4 /WX /utf-8)
add_executable(audio_mixer_test MixerTest.cpp Mixer.cpp)
target_compile_features(audio_mixer_test PRIVATE cxx_std_20)
target_compile_options(audio_mixer_test PRIVATE /W4 /WX /utf-8)
enable_testing()
add_test(NAME audio_mixer COMMAND audio_mixer_test)

# Players launch this GUI-subsystem wrapper. The EVEngine host remains a
# console-subsystem executable but is started with CREATE_NO_WINDOW.
add_executable(SuperMario WIN32 Launcher.cpp Launcher.rc)
target_include_directories(SuperMario PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(SuperMario PRIVATE cxx_std_20)
target_compile_options(SuperMario PRIVATE /W4 /WX /utf-8)
target_link_libraries(SuperMario PRIVATE user32)
